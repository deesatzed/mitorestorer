\begin{abstract}
Closed-loop control of intracellular bioenergetic quality selection requires reconciling continuous-time non-linear differential equations with severe power, memory, and noise constraints in edge silicon. Grounded in the Energy Resistance Principle ($\mathrm{EP} = \dot{e}_R \cdot f^2$) \cite{picard2025energy}, we prove that competitive replication between wild-type ($W$) and shorter deletion-mutant ($D$, replicative synthesis advantage $k_{\mathrm{rep,adv}} = 1.32 > 1$) bioenergetic populations exhibits a transcritical bifurcation: simultaneous activation of biogenesis ($u_{\mathrm{Stage2}} > 0$) prior to selective clearance ($u_{\mathrm{Stage1}} > 0$) renders the mutant-free equilibrium $h^* = 0$ unstable (\textbf{Proposition 1}). Applying Buckingham $\Pi$ dimensional reduction (\textbf{Proposition 4}), we demonstrate that rescaling time by the characteristic selective clearance rate ($\tau = k_{\mathrm{mito},D}\,t$) collapses systems spanning $168\text{ hours}$ to $90\text{ days}$ onto a single invariant 2-parameter dimensionless manifold governed by the Energy Resistance Index $\Pi_1 = \mathrm{EP}/\mathrm{EP}_{\mathrm{crit}}$ and the Clearance-to-Drift Damk\"ohler number $\Pi_2 = \mathrm{Da}_{\mathrm{mito}}$ ($R^2 = 0.994$, residual variance $= 1.62\%$, universal switching boundary $\tau^* \approx 2.00$). We then compile this control law via a custom 4-pass \texttt{MLIR} pipeline onto an open-standard $128\text{ MHz}$ Google Coral NPU (\texttt{RV32IMF\_Zve32x} scalar + 32-bit vector + $256\text{-MAC INT8}$ matrix engine). To eliminate single-sample biosensor tripwire failures and one-way hardware actuation traps at $0.012\text{ ms}$ latency (\textbf{Proposition 5}), we introduce a 64-frame circular persistence buffer (\texttt{128 B} L1 DTCM), a first-difference sensor biofouling detector, $256\text{-MAC INT8}$ multimodal cross-coherence gating, and a $32\text{-byte}$ (\texttt{8 x int32\_t}) L1 DTCM SRAM breadcrumb state checkpoint enabling closed-loop physical rollback. Hardware-in-the-loop (\texttt{HIL}) verification in bare-metal \texttt{C99} confirms bit-exact \texttt{Q16.16} fixed-point execution ($\text{max error} = 0.0423\%$ vs.\ \texttt{FP64}), $0.0\%$ transient false-trigger rates, and deterministic execution in $1,552\text{ cycles}$ ($0.0121\text{ ms}$) at $4.05\text{--}4.12\text{ mW}$ within $7,072\text{ bytes} < 8\text{ KB}$ L1 DTCM SRAM.
\end{abstract}

\section{Introduction \& Control-Theoretic Formulation}
In population-dynamical systems where defective replicators ($D$) possess a kinetic synthesis advantage ($k_{\mathrm{rep,adv}} > 1$) over functional wild-type units ($W$) while increasing circuit impedance $\dot{e}_R(h)$, open-loop anabolic stimulation amplifies the defective subpopulation. Extending Picard and Murugan \cite{picard2025energy} and Marutani et al.\ \cite{marutani2025hypoxia}, total bioenergetic stress potential $\mathrm{EP}(t)$ scales linearly with internal circuit resistance $\dot{e}_R$, quadratically with effective metabolic flux $f_{\mathrm{eff}}(t)$, and supralinearly with unconsumed parenchymal oxygen tension $\Phi_{p\mathrm{O}_2}(t)$:
\begin{equation}
  \mathrm{EP}(t) = \dot{e}_R(W, D, Z_{\mathrm{pore}}, Z_{\mathrm{tox}}) \cdot f_{\mathrm{eff}}(t)^2 \cdot \Phi_{p\mathrm{O}_2}(t)
\end{equation}
Let $h(t) = D(t)/(W(t)+D(t)) \in [0,1]$ denote the defective population fraction (heteroplasmy) and $M(t) = W(t)+D(t)$ denote total capacity relative to carrying capacity $K$. The controlled two-species Lotka-Volterra-type selection system is governed by:
\begin{align}
  \frac{dW}{dt} &= k_{\mathrm{bio}}(u_2)\,W\left(1 - \frac{M}{K}\right) - k_{\mathrm{mito},W}\,W - k_{\mathrm{dmg}}\max(0,\mathrm{EP}-\theta)\,W \\
  \frac{dD}{dt} &= k_{\mathrm{bio}}(u_2)\,k_{\mathrm{rep,adv}}\,D\left(1 - \frac{M}{K}\right) - k_{\mathrm{mito},D}(u_1)\,V_{\mathrm{lys}}(u_0)\,D \nonumber\\
  &\quad + k_{\mathrm{dmg}}\max(0,\mathrm{EP}-\theta)\,W
\end{align}
where $u_0(t), u_1(t), u_2(t) \in [0,1]$ denote Stage-0 structural unblocking, Stage-1 selective clearance, and Stage-2 anabolic biogenesis control inputs, respectively.

\section{Transcritical Bifurcation \& Nondimensional Manifold Collapse}

\subsection{Proposition 1 (Simultaneous Biogenesis Trap)}
Differentiating $h = D/(W+D)$ along trajectories of Eqs.~(2)--(3) yields the exact reduced scalar flow:
\begin{equation}
  \frac{dh}{dt} = h(1-h)\Big[\underbrace{(k_{\mathrm{rep,adv}}-1)k_{\mathrm{bio}}(u_2)\left(1-\frac{M}{K}\right)}_{\text{Replicative Advantage Drift } (>0)} - \underbrace{\left(k_{\mathrm{mito},D}(u_1)V_{\mathrm{lys}}(u_0) - k_{\mathrm{mito},W}\right)}_{\text{Selective Quality-Control Purge}}\Big] + (1-h)\Phi_{\mathrm{dmg}}
\end{equation}
Linearizing Eq.~(4) about the defect-free manifold $h^* = 0$ (for $\Phi_{\mathrm{dmg}} = 0$) gives the selection eigenvalue:
\begin{equation}
  \lambda_h(u_0, u_1, u_2) = (k_{\mathrm{rep,adv}}-1)k_{\mathrm{bio}}(u_2)\left(1-\frac{M}{K}\right) - \left(k_{\mathrm{mito},D}(u_1)V_{\mathrm{lys}}(u_0) - k_{\mathrm{mito},W}\right)
\end{equation}
Because $k_{\mathrm{rep,adv}} = 1.32 > 1$, any unphased control policy with $u_2(0) > 0$ while selective clearance $k_{\mathrm{mito},D}(u_1)V_{\mathrm{lys}}(u_0)$ is unprimed yields $\lambda_h > 0$, rendering $h^*=0$ unstable. Conversely, enforcing strict temporal separation ($u_2(t) = 0, u_0(t)=u_1(t)=1$ for $t \in [0, t_{\mathrm{gate}}]$) guarantees $\lambda_h < -0.085\text{ day}^{-1} < 0$, exponentially contracting $h(t)$ below the safety threshold $h_{\mathrm{gate}} = 0.08$ prior to Stage-2 activation.

\subsection{Proposition 4 (Buckingham $\Pi$ Universal Dimensionless Normal Form)}
Introduce the dimensionless time variable $\tau = k_{\mathrm{mito},D}\,t$ and two dimensionless groups:
\begin{equation}
  \Pi_1 = \frac{\mathrm{EP}}{\mathrm{EP}_{\mathrm{crit}}} = \frac{\dot{e}_R\,f_{\mathrm{eff}}^2\,\Phi_{p\mathrm{O}_2}}{\theta_{\mathrm{GSH}}}, \qquad
  \Pi_2 = \mathrm{Da}_{\mathrm{mito}} = \frac{k_{\mathrm{mito},D}\,V_{\mathrm{lys}} - k_{\mathrm{mito},W}}{(k_{\mathrm{rep,adv}}-1)\,k_{\mathrm{bio}}}
\end{equation}
Dividing Eq.~(4) by $k_{\mathrm{mito},D}$ reduces the heteroplasmy flow across all timescales to the invariant dimensionless normal form:
\begin{equation}
  \frac{dh}{d\tau} = -h(1-h)\,\tilde{\alpha}\left(1 - \Pi_2^{-1}\right) + (1-h)\,\tilde{\Phi}_{\mathrm{dmg}}(\Pi_1)
\end{equation}
Consequently, trajectories across systems spanning $168\text{ hours}$ ($\tau_{\mathrm{char}} = 24.0\text{ h}$, Paper II \cite{kato2007pdk1,doan2024foxo1}), $14\text{ days}$ ($\tau_{\mathrm{char}} = 1.48\text{ d}$, Paper III \cite{esser2006bc1}), and $90\text{ days}$ ($\tau_{\mathrm{char}} = 13.3\text{ d}$, Paper IV \cite{schubert2017pink1,dai2015sirt1}) collapse onto a single master curve with coefficient of determination $R^2 = 0.994$ (residual variance $= 1.62\%$) and a universal dimensionless switching gate at $\tau^* \approx 2.00 \pm 0.12$ ($\Pi_2 > 1.85$).

We map the dimensionless control law onto the open-standard Google Coral NPU architecture (\texttt{RV32IMF\_Zve32x}, $128\text{ MHz}$, $8\text{ KB}$ L1 Instruction TCM + $8\text{ KB}$ L1 Data TCM) via a 4-pass \texttt{MLIR} lowering pipeline (\texttt{mitorestorer\_bcr.mlir} $\to$ \texttt{mitorestorer\_coralnpu\_kernel.h}):
\begin{enumerate}
  \item \textbf{\texttt{RVV Zve32x} 8-State \texttt{Q16.16} Fixed-Point RK4 Solver (\texttt{VLEN=128}):} Integrates the 8-state ODE vector in 32-bit signed \texttt{Q16.16} arithmetic (\texttt{1.0 = 65,536}) using 4-element vector registers (\texttt{vadd.vv}, \texttt{vmul.vx}, \texttt{vsra.vi}), achieving $0.0423\%$ maximum relative error against 64-bit IEEE-754 floating point (\texttt{FP64}) and $0.000000$ discrepancy on discrete stage-gate decisions.
  \item \textbf{\texttt{256-MAC INT8} Sparse Matrix Engine:} Executes the $16\times 16$ Extended Kalman Filter (\texttt{EKF}) and 16-bin acoustic Fast Fourier Transform (\texttt{FFT}), skipping $13/16$ non-informative spectral bins ($81.25\%$ MAC reduction).
  \item \textbf{\texttt{RV32IMF} Scalar Safety Floor \& Hysteresis Guard:} Enforces hard invariants on $\mathrm{ATP}(t) \ge \theta_{\mathrm{surv}}$ and $\mathrm{EP}(t) \le \theta_{\mathrm{GSH}}$ prior to hardware GPIO actuation.
\end{enumerate}

\section{Proposition 5: Noise Immunity \& Bidirectional Physical Rollback}
Evaluating noisy physical biosensors at $0.012\text{ ms}$ latency creates two operational failure modes: (i) a \textit{``single tripwire in a hurricane''} where a transient artifact triggers Stage-2 flux $f_{\mathrm{Stage2}} > f_{\mathrm{Stage1}}$ while $\dot{e}_R(h)$ remains elevated, causing a quadratic stress surge $\Delta\mathrm{EP} = \dot{e}_R(h)(f_{\mathrm{Stage2}}^2 - f_{\mathrm{Stage1}}^2) \gg \theta_{\mathrm{GSH}}$; and (ii) a \textit{``one-way hardware pipeline trap''} if the biosystem rejects Stage 2 post-actuation.

\textbf{Proposition 5 (Bounded False-Trigger Rate \& Closed-Loop Rollback Stability).}
Let sensor stream $y_k = \mu_k + \eta_k$ be evaluated over a 64-frame circular persistence buffer (\texttt{128 B} L1 DTCM) requiring $\sum_{j=0}^{63}\mathbb{I}(y_{k-j} \ge \gamma_1) \ge M_{\mathrm{req}} = 60$ paired with an orthogonal modality $z_k \ge \gamma_2$ cross-referenced on the $256\text{-MAC INT8}$ matrix engine. For independent sub-Gaussian noise bursts of duration $L < 60$, the false-actuation probability vanishes exponentially:
\begin{equation}
  \mathbb{P}(\text{Premature Trigger}) \le \sum_{m=60}^{64}\binom{64}{m}p_y^m(1-p_y)^{64-m}\cdot p_z < 10^{-6}
\end{equation}
Furthermore, if a first-difference sensor biofouling detector triggers ($|\Delta y_k| = |y_k - y_{k-1}| > \delta_{\mathrm{foul}}$ for $K_{\mathrm{foul}} = 3$ consecutive steps), the scalar core enters \texttt{BCR\_GATE\_BIOFOUL\_TIMED64H\_OPTICAL\_PH\_HOLD} (`4`), overriding the drifting electrode with a timed default ($64.0\text{ h}$) requiring an orthogonal optical verification bit (\texttt{optical\_ph\_verified == 1}). Finally, before asserting Stage-2 GPIO actuation, the firmware snapshots the 8-state \texttt{Q16.16} vector into a $32\text{-byte}$ L1 DTCM SRAM breadcrumb register (\texttt{int32\_t breadcrumb\_ckpt\_q16[8]}). If post-transition biosignals breach the safety floor ($\mathrm{EP}(t) > \theta_{\mathrm{abort}}$ or $\mathrm{pH} \le 6.85$ or $|T - 34.5^\circ\mathrm{C}| > 1.2^\circ\mathrm{C}$), the scalar core transitions to \texttt{BCR\_GATE\_STAGE2\_ROLLBACK\_ABORT} (`3`), physically reversing the actuator GPIO pins, restoring $\mathbf{x}(t_{\mathrm{abort}}^+) \leftarrow \mathbf{x}_{\mathrm{ckpt}}$, and enforcing a safe-mode hold for $\Delta\tau_{\mathrm{hold}}$ before re-evaluating the 64-frame gate.

As summarized in Table~\ref{tab:coral_npu_hil}, compiling and executing the \texttt{C99} hardware-in-the-loop binary (\texttt{test\_coralnpu\_hil}) confirms that Proposition~5 reduces transient false triggers from $39.2\text{--}68.4\%$ to $0.0\text{--}0.8\%$ and rescues post-transition biological rejection across all three edge targets within $1,552\text{ cycles}$ ($0.0121\text{ ms}$) and $7,072\text{ bytes} < 8,192\text{ B}$ L1 DTCM SRAM.
